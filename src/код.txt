    $regResp = Invoke-WebRequest http://localhost:3000/auth/register -Method POST -ContentType "application/json" -Body '{"username":"demo_student","email":"demo@psu.by","password":"secret123"}'
    Write-Host $regResp.Content
} catch { 
    Write-Host "Ошибка (возможно, пользователь уже создан): $($_.ErrorDetails.Message)" -ForegroundColor Red 
}

# 3. ДОСТУП БЕЗ ТОКЕНА (ЛР 17: JwtAuthGuard -> 401)
try { 
    Invoke-WebRequest http://localhost:3000/tasks 
} catch { 
    Write-Host "Статус: $($_.Exception.Response.StatusCode.value__) (Успех! Доступ закрыт)" -ForegroundColor Red 
    Write-Host $_.ErrorDetails.Message 
}
